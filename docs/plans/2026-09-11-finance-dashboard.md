# pipeline/utils.py
"""Formatting and utility helpers for the finance pipeline."""

def fmt_millions(value: float) -> float:
    """Convert raw NGN to millions, rounded to 1 decimal."""
    return round(value / 1_000_000, 1)

def fmt_pct(value: float) -> str:
    """Format a ratio as percentage string."""
    return f"{value:.1f}%"

def fmt_ratio(value: float) -> str:
    """Format a ratio with × suffix."""
    return f"{value:.2f}×"

def safe_div(numerator: float, denominator: float, default: float = 0.0) -> float:
    """Safe division returning default if denominator is zero."""
    if denominator == 0:
        return default
    return numerator / denominator
```

# pipeline/read_data.py
"""Reads GL_Clean.xlsx and Statement_Mapping.xlsx into DataFrames."""

import pandas as pd
from pathlib import Path


def read_gl_clean(path: str | Path) -> pd.DataFrame:
    """Read the GL_Clean sheet from the cleaned GL workbook."""
    df = pd.read_excel(path, sheet_name="GL_Clean")
    # Ensure numeric columns
    for col in ["Debit", "Credit", "Net"]:
        df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0.0)
    # Ensure date columns
    df["Doc_Date"] = pd.to_datetime(df["Doc_Date"], errors="coerce")
    df["Month_No"] = pd.to_numeric(df["Month_No"], errors="coerce").astype("Int64")
    df["Year"] = pd.to_numeric(df["Year"], errors="coerce").astype("Int64")
    return df


def read_statement_mapping(path: str | Path) -> pd.DataFrame:
    """Read the Statement Mapping sheet."""
    df = pd.read_excel(path, sheet_name="Statement Mapping")
    df["GL_Code"] = df["GL_Code"].astype(str).str.strip()
    return df


def read_account_summary(path: str | Path) -> pd.DataFrame:
    """Read the Account_Summary sheet (opening balances, source totals)."""
    df = pd.read_excel(path, sheet_name="Account_Summary")
    df["GL_Code"] = df["GL_Code"].astype(str).str.strip()
    for col in ["Opening_Debit", "Opening_Credit", "Src_Total_Debit",
                "Src_Total_Credit", "Src_Ending_Net"]:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0.0)
    return df
```

# pipeline/build_statements.py
"""Aggregates GL transactions into P&L, Balance Sheet, and Cash Flow."""

import pandas as pd
from .utils import safe_div


def merge_mapping(gl: pd.DataFrame, mapping: pd.DataFrame) -> pd.DataFrame:
    """Left-join GL transactions with statement mapping."""
    gl["GL_Code"] = gl["GL_Code"].astype(str).str.strip()
    merged = gl.merge(mapping, on="GL_Code", how="left")
    merged["Statement_Section"] = merged["Statement_Section"].fillna("Unclassified")
    merged["FS_Heading"] = merged["FS_Heading"].fillna("Unclassified")
    merged["Note_Heading"] = merged["Note_Heading"].fillna("Unclassified")
    merged["CF_Category"] = merged["CF_Category"].fillna("Unclassified")
    merged["Segment"] = merged["Segment"].fillna("Corporate")
    return merged


def build_pl(merged: pd.DataFrame, period: str | None = None) -> dict:
    """Build P&L from merged GL data. If period is given, filter to that period."""
    df = merged[merged["Statement_Section"].isin(["Revenue", "Cost of Sales", "Operating Expenses", "Other Income", "Finance Costs", "Tax"])].copy()
    if period:
        df = df[df["Period"] == period]

    # Group by FS_Heading, sum Debit and Credit, compute Net
    grouped = df.groupby(["Statement_Section", "FS_Heading"]).agg(
        Debit=("Debit", "sum"),
        Credit=("Credit", "sum"),
        Net=("Net", "sum"),
    ).reset_index()

    # For P&L: Revenue is Credit - Debit (income), Expenses are Debit - Credit (cost)
    pl_lines = []
    for _, row in grouped.iterrows():
        section = row["Statement_Section"]
        heading = row["FS_Heading"]
        if section in ["Revenue", "Other Income"]:
            amount = row["Credit"] - row["Debit"]  # Income = credits - debits
        else:
            amount = row["Debit"] - row["Credit"]  # Expense = debits - credits
        pl_lines.append({
            "section": section,
            "heading": heading,
            "amount": round(amount, 2),
        })

    # Build structured P&L
    revenue = sum(l["amount"] for l in pl_lines if l["section"] == "Revenue")
    cogs = sum(l["amount"] for l in pl_lines if l["section"] == "Cost of Sales")
    gp = revenue - cogs
    opex = sum(l["amount"] for l in pl_lines if l["section"] == "Operating Expenses")
    other_income = sum(l["amount"] for l in pl_lines if l["section"] == "Other Income")
    finance_costs = sum(l["amount"] for l in pl_lines if l["section"] == "Finance Costs")
    tax = sum(l["amount"] for l in pl_lines if l["section"] == "Tax")

    ebit = gp - opex + other_income
    pbt = ebit - finance_costs
    pat = pbt - tax

    return {
        "revenue": round(revenue, 2),
        "cogs": round(cogs, 2),
        "gross_profit": round(gp, 2),
        "operating_expenses": round(opex, 2),
        "other_income": round(other_income, 2),
        "finance_costs": round(finance_costs, 2),
        "tax": round(tax, 2),
        "ebit": round(ebit, 2),
        "pbt": round(pbt, 2),
        "pat": round(pat, 2),
        "gp_margin": round(safe_div(gp, revenue) * 100, 1),
        "op_margin": round(safe_div(ebit, revenue) * 100, 1),
        "net_margin": round(safe_div(pat, revenue) * 100, 1),
        "line_items": pl_lines,
    }


def build_balance_sheet(merged: pd.DataFrame, account_summary: pd.DataFrame, period: str | None = None) -> dict:
    """Build Balance Sheet from GL data and account summary."""
    df = merged[merged["Statement_Section"].isin(["Assets", "Liabilities", "Equity"])].copy()
    if period:
        df = df[df["Period"] == period]

    # For balance sheet, use closing balances from account summary
    # Net balance = Opening_Debit - Opening_Credit + (Txn_Debit - Txn_Credit)
    # We'll approximate from the merged data: sum of Net for asset/liability codes
    grouped = df.groupby(["Statement_Section", "FS_Heading"]).agg(
        Net=("Net", "sum"),
    ).reset_index()

    # Get opening balances from account_summary
    bs_codes = df["GL_Code"].unique()
    acct_balances = account_summary[account_summary["GL_Code"].isin(bs_codes)].copy()

    # Build structured B/S
    total_assets = 0
    total_liabilities = 0
    total_equity = 0
    bs_lines = []

    for _, row in grouped.iterrows():
        section = row["Statement_Section"]
        heading = row["FS_Heading"]
        # For assets: Net represents movement; closing = opening + movement
        # We'll use the Net from transactions as the period movement
        amount = row["Net"]
        bs_lines.append({
            "section": section,
            "heading": heading,
            "amount": round(amount, 2),
        })
        if section == "Assets":
            total_assets += amount
        elif section == "Liabilities":
            total_liabilities += amount
        elif section == "Equity":
            total_equity += amount

    return {
        "total_assets": round(total_assets, 2),
        "total_liabilities": round(total_liabilities, 2),
        "total_equity": round(total_equity, 2),
        "line_items": bs_lines,
    }


def build_cash_flow(merged: pd.DataFrame, period: str | None = None) -> dict:
    """Build Cash Flow from GL data using CF_Category mapping."""
    df = merged[merged["CF_Category"].isin(["Operating", "Investing", "Financing"])].copy()
    if period:
        df = df[df["Period"] == period]

    grouped = df.groupby(["CF_Category", "FS_Heading"]).agg(
        Net=("Net", "sum"),
    ).reset_index()

    cf_sections = {}
    for cat in ["Operating", "Investing", "Financing"]:
        cat_data = grouped[grouped["CF_Category"] == cat]
        items = [{"heading": r["FS_Heading"], "amount": round(r["Net"], 2)}
                 for _, r in cat_data.iterrows()]
        total = sum(i["amount"] for i in items)
        cf_sections[cat.lower()] = {
            "items": items,
            "total": round(total, 2),
        }

    net_movement = sum(cf_sections[c]["total"] for c in cf_sections)

    return {
        "operating": cf_sections.get("operating", {"items": [], "total": 0}),
        "investing": cf_sections.get("investing", {"items": [], "total": 0}),
        "financing": cf_sections.get("financing", {"items": [], "total": 0}),
        "net_movement": round(net_movement, 2),
    }


def build_segments(merged: pd.DataFrame, period: str | None = None) -> list:
    """Build revenue by segment breakdown."""
    df = merged[merged["Statement_Section"] == "Revenue"].copy()
    if period:
        df = df[df["Period"] == period]

    grouped = df.groupby("Segment").agg(
        Revenue=("Credit", "sum"),
        COGS=("Debit", "sum"),
    ).reset_index()

    segments = []
    for _, row in grouped.iterrows():
        rev = row["Revenue"] - row.get("COGS", 0)  # Net revenue
        segments.append({
            "segment": row["Segment"],
            "revenue": round(max(rev, 0), 2),
            "cost": round(row.get("COGS", 0), 2),
        })

    return segments


def build_monthly_summary(merged: pd.DataFrame) -> list:
    """Build monthly P&L summary for trend charts."""
    df = merged[merged["Statement_Section"].isin(["Revenue", "Cost of Sales", "Operating Expenses"])].copy()

    monthly = df.groupby(["Period", "Statement_Section"]).agg(
        Debit=("Debit", "sum"),
        Credit=("Credit", "sum"),
    ).reset_index()

    results = []
    for period in sorted(monthly["Period"].unique()):
        period_data = monthly[monthly["Period"] == period]
        rev_row = period_data[period_data["Statement_Section"] == "Revenue"]
        cogs_row = period_data[period_data["Statement_Section"] == "Cost of Sales"]
        opex_row = period_data[period_data["Statement_Section"] == "Operating Expenses"]

        revenue = float(rev_row["Credit"].sum() - rev_row["Debit"].sum()) if len(rev_row) else 0
        cogs = float(cogs_row["Debit"].sum() - cogs_row["Credit"].sum()) if len(cogs_row) else 0
        opex = float(opex_row["Debit"].sum() - opex_row["Credit"].sum()) if len(opex_row) else 0

        results.append({
            "period": period,
            "revenue": round(revenue, 2),
            "cogs": round(cogs, 2),
            "gross_profit": round(revenue - cogs, 2),
            "operating_expenses": round(opex, 2),
            "operating_profit": round(revenue - cogs - opex, 2),
        })

    return results
```

"""Calculates financial ratios from statement data."""

from .utils import safe_div


def calculate_ratios(pl: dict, bs: dict, cf: dict) -> dict:
    """Calculate key financial ratios from statements."""
    revenue = pl["revenue"]
    gp = pl["gross_profit"]
    ebit = pl["ebit"]
    pat = pl["pat"]
    total_assets = bs["total_assets"]
    total_equity = bs["total_equity"]
    total_liabilities = bs["total_liabilities"]
    operating_cf = cf["operating"]["total"]

    return {
        "profitability": {
            "gp_margin": {"value": pl["gp_margin"], "label": "Gross margin", "basis": "GP ÷ revenue"},
            "op_margin": {"value": pl["op_margin"], "label": "Operating margin", "basis": "EBIT ÷ revenue"},
            "net_margin": {"value": pl["net_margin"], "label": "Net margin", "basis": "PAT ÷ revenue"},
            "opex_ratio": {"value": round(safe_div(pl["operating_expenses"], revenue) * 100, 1), "label": "Opex ratio", "basis": "Opex ÷ revenue"},
            "roa": {"value": round(safe_div(pat, total_assets) * 100, 1), "label": "Return on assets", "basis": "PAT ÷ total assets"},
            "roe": {"value": round(safe_div(pat, total_equity) * 100, 1), "label": "Return on equity", "basis": "PAT ÷ equity"},
        },
        "liquidity": {
            "current_ratio": {"value": round(safe_div(total_assets, total_liabilities), 2), "label": "Current ratio", "basis": "Total assets ÷ total liabilities"},
            "cash_to_liabilities": {"value": round(safe_div(cf["operating"]["total"], total_liabilities), 2), "label": "Cash flow ratio", "basis": "Operating CF ÷ liabilities"},
        },
        "leverage": {
            "debt_to_equity": {"value": round(safe_div(total_liabilities, total_equity) * 100, 1), "label": "Debt to equity", "basis": "Liabilities ÷ equity"},
            "debt_to_assets": {"value": round(safe_div(total_liabilities, total_assets) * 100, 1), "label": "Debt to assets", "basis": "Liabilities ÷ total assets"},
        },
        "efficiency": {
            "asset_turnover": {"value": round(safe_div(revenue, total_assets), 2), "label": "Asset turnover", "basis": "Revenue ÷ total assets"},
            "cash_conversion": {"value": round(safe_div(operating_cf, ebit) * 100, 1) if ebit > 0 else 0, "label": "Cash conversion", "basis": "Operating CF ÷ EBIT"},
        },
    }

# pipeline/commentary.py
"""Auto-generates narrative commentary from financial figures."""

from .utils import safe_div


def generate_commentary(pl: dict, bs: dict, cf: dict, ratios: dict, monthly: list) -> dict:
    """Generate structured commentary sections from financial data."""
    sections = {}

    # Revenue commentary
    revenue = pl["revenue"]
    if monthly and len(monthly) >= 2:
        prior = monthly[-2]["revenue"]
        growth = round(safe_div(revenue - prior, prior) * 100, 1) if prior else 0
        trend = "up" if growth > 0 else "down" if growth < 0 else "flat"
        sections["revenue"] = {
            "narrative": f"Revenue for the period was ₦{revenue/1e9:.2f}B, {abs(growth)}% {trend} vs prior month (₦{prior/1e9:.2f}B).",
            "current": revenue,
            "prior": prior,
            "growth_pct": growth,
        }
    else:
        sections["revenue"] = {
            "narrative": f"Revenue for the period was ₦{revenue/1e9:.2f}B.",
            "current": revenue,
        }

    # Cost commentary
    cogs_pct = pl["gp_margin"]
    sections["costs"] = {
        "narrative": f"COGS consumed {100 - cogs_pct:.1f}% of revenue. Gross margin was {cogs_pct}%. Operating expenses were ₦{pl['operating_expenses']/1e6:,.0f}M.",
        "cogs_pct_revenue": round(100 - cogs_pct, 1),
        "gp_margin": cogs_pct,
    }

    # Profitability
    sections["profitability"] = {
        "narrative": f"Gross profit margin {pl['gp_margin']}%, operating margin {pl['op_margin']}%, net margin {pl['net_margin']}%.",
        "gp_margin": pl["gp_margin"],
        "op_margin": pl["op_margin"],
        "net_margin": pl["net_margin"],
    }

    # Cash flow
    op_cf = cf["operating"]["total"]
    inv_cf = cf["investing"]["total"]
    fin_cf = cf["financing"]["total"]
    sections["cash"] = {
        "narrative": f"Net cash movement ₦{cf['net_movement']/1e6:,.0f}M. Operating ₦{op_cf/1e6:,.0f}M, investing ₦{inv_cf/1e6:,.0f}M, financing ₦{fin_cf/1e6:,.0f}M.",
        "net_movement": cf["net_movement"],
        "operating": op_cf,
        "investing": inv_cf,
        "financing": fin_cf,
    }

    # Ratios summary
    cr = ratios["liquidity"]["current_ratio"]["value"]
    de = ratios["leverage"]["debt_to_equity"]["value"]
    sections["ratios"] = {
        "narrative": f"Current ratio {cr}×, debt-to-equity {de}%, asset turnover {ratios['efficiency']['asset_turnover']['value']}×.",
        "current_ratio": cr,
        "debt_to_equity": de,
    }

    # Things to watch (auto-generated flags)
    flags = []
    if pl["gp_margin"] < 15:
        flags.append({"type": "warning", "text": f"Gross margin {pl['gp_margin']}% is below 15% threshold"})
    if pl["net_margin"] < 0:
        flags.append({"type": "danger", "text": "Company is loss-making for this period"})
    if cr < 1.0:
        flags.append({"type": "warning", "text": f"Current ratio {cr}× below 1.0 — potential liquidity concern"})
    if de > 200:
        flags.append({"type": "warning", "text": f"Debt-to-equity {de}% is elevated"})

    if not flags:
        flags.append({"type": "success", "text": "No material concerns detected in the financial statements"})

    sections["things_to_watch"] = flags

    return sections
```

sys.path.insert(0, str(Path(__file__).parent.parent))

from pipeline.read_data import read_gl_clean, read_statement_mapping, read_account_summary
from pipeline.build_statements import (
    merge_mapping, build_pl, build_balance_sheet, build_cash_flow,
    build_segments, build_monthly_summary,
)
from pipeline.ratios import calculate_ratios
from pipeline.commentary import generate_commentary


def run_pipeline(
    gl_path: str = "Sample GL_clean.xlsx",
    mapping_path: str = "Statement_Mapping.xlsx",
    output_path: str = "dashboard/public/data/dashboard_data.json",
    period: str | None = None,
):
    """Run the full pipeline and write dashboard_data.json."""
    base = Path(__file__).parent.parent

    # 1. Read data
    print("Reading GL data...")
    gl = read_gl_clean(base / gl_path)
    sm = read_statement_mapping(base / mapping_path)
    ac = read_account_summary(base / gl_path)
    print(f"  GL: {len(gl)} transactions, {gl['GL_Code'].nunique()} accounts")
    print(f"  Mapping: {len(sm)} codes")

    # 2. Merge
    print("Merging with statement mapping...")
    merged = merge_mapping(gl, sm)
    unmapped = merged[merged["Statement_Section"] == "Unclassified"]
    if len(unmapped) > 0:
        print(f"  WARNING: {len(unmapped)} transactions unmapped (GL codes: {unmapped['GL_Code'].unique().tolist()})")

    # 3. Build statements
    print("Building financial statements...")
    pl = build_pl(merged, period)
    bs = build_balance_sheet(merged, ac, period)
    cf = build_cash_flow(merged, period)
    segments = build_segments(merged, period)
    monthly = build_monthly_summary(merged)

    print(f"  Revenue: ₦{pl['revenue']/1e9:.2f}B")
    print(f"  GP margin: {pl['gp_margin']}%")
    print(f"  Net margin: {pl['net_margin']}%")

    # 4. Calculate ratios
    print("Calculating ratios...")
    ratios = calculate_ratios(pl, bs, cf)

    # 5. Generate commentary
    print("Generating commentary...")
    commentary = generate_commentary(pl, bs, cf, ratios, monthly)

    # 6. Assemble output
    dashboard_data = {
        "period": period or "Full Year 2025",
        "company": "CIG Motors",
        "currency": "NGN",
        "unit": "millions",
        "generated_at": __import__("datetime").datetime.now().isoformat(),
        "data_sources": {
            "gl_transactions": len(gl),
            "gl_accounts": int(gl["GL_Code"].nunique()),
            "mapping_codes": int(len(sm)),
            "unmapped_transactions": int(len(unmapped)),
        },
        "pl": pl,
        "balance_sheet": bs,
        "cash_flow": cf,
        "segments": segments,
        "monthly": monthly,
        "ratios": ratios,
        "commentary": commentary,
        "available_data": {
            "budget": False,
            "covenants": False,
            "prior_period": False,
            "commentary_manual": False,
        },
    }

    # 7. Write JSON
    out = base / output_path
    out.parent.mkdir(parents=True, exist_ok=True)
    with open(out, "w", encoding="utf-8") as f:
        json.dump(dashboard_data, f, indent=2, default=str)
    print(f"\n✓ Dashboard data written to {out}")
    print(f"  File size: {out.stat().st_size / 1024:.1f} KB")

    return dashboard_data


if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Generate dashboard data from GL")
    parser.add_argument("--gl", default="Sample GL_clean.xlsx", help="Path to GL_Clean workbook")
    parser.add_argument("--mapping", default="Statement_Mapping.xlsx", help="Path to Statement_Mapping")
    parser.add_argument("--output", default="dashboard/public/data/dashboard_data.json", help="Output JSON path")
    parser.add_argument("--period", default=None, help="Filter to specific period (e.g. 2025-08)")
    args = parser.parse_args()
    run_pipeline(args.gl, args.mapping, args.output, args.period)

// src/lib/format.js
export function fmtM(n) {
  if (n == null || isNaN(n)) return '—'
  const abs = Math.abs(n)
  if (abs >= 1e9) return `${n < 0 ? '(' : ''}${(abs / 1e9).toFixed(2)}B${n < 0 ? ')' : ''}`
  if (abs >= 1e6) return `${n < 0 ? '(' : ''}${(abs / 1e6).toFixed(0)}M${n < 0 ? ')' : ''}`
  if (abs >= 1e3) return `${n < 0 ? '(' : ''}${(abs / 1e3).toFixed(0)}K${n < 0 ? ')' : ''}`
  return n.toLocaleString()
}

export function fmtFull(n) {
  if (n == null || isNaN(n)) return '—'
  return n.toLocaleString('en-NG', { minimumFractionDigits: 0, maximumFractionDigits: 0 })
}

export function fmtPct(n) {
  if (n == null || isNaN(n)) return '—'
  return `${n.toFixed(1)}%`
}

export function fmtRatio(n) {
  if (n == null || isNaN(n)) return '—'
  return `${n.toFixed(2)}×`
}

export function fmtSigned(n) {
  if (n == null || isNaN(n)) return '—'
  const prefix = n > 0 ? '+' : n < 0 ? '(' : ''
  const suffix = n < 0 ? ')' : ''
  return `${prefix}${fmtFull(Math.abs(n))}${suffix}`
}
